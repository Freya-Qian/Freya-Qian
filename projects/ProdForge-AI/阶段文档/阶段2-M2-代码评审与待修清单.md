## 三、修法细节（按文件）

### `services/positioning.py`（#1 无证据结论清单）
```python
def _clean_differentiators(raw: list, valid_ref_keys: set) -> list[dict]:
    result = []
    for item in raw or []:
        if not isinstance(item, dict):
            continue
        point = str(item.get("point", "")).strip()
        if not point:
            continue
        refs = [r for r in (item.get("evidence_refs") or []) if r in valid_ref_keys]
        if not refs:
            point = point + "（待补证：无证据引用）"   # 无证据支撑的差异点显式标注
        result.append({"point": point, "evidence_refs": refs})
    return result
```

### `api/routes.py`（#2 证据卡修改）
```python
@router.patch("/projects/{project_id}/competitors/{competitor_id}/evidences/{evidence_id}",
              response_model=EvidenceOut)
def update_evidence(project_id: str, competitor_id: str, evidence_id: str,
                    payload: EvidenceUpdate, db: Session = Depends(get_db)):
    _require_competitor(db, project_id, competitor_id)
    ev = db.get(CompetitorEvidence, evidence_id)
    if ev is None or ev.competitor_id != competitor_id:
        raise LLMError("NOT_FOUND", "证据卡不存在")
    for field in ("structured_claim", "implication", "confidence",
                  "evidence_type", "source_type", "source_url", "raw_excerpt"):
        if getattr(payload, field, None) is not None:
            setattr(ev, field, getattr(payload, field))
    if payload.source_url is not None:
        ev.source_url = competitor_service.validate_source_url(payload.source_url)
    if payload.confidence is not None and payload.confidence not in CONFIDENCE_LEVELS:
        raise LLMError("INVALID_INPUT", "confidence 不合法")
    if payload.evidence_type is not None and payload.evidence_type not in EVIDENCE_TYPES:
        raise LLMError("INVALID_INPUT", "evidence_type 不合法")
    db.commit()
    db.refresh(ev)
    return ev
```

### `api/routes.py`（#5 悬空引用）
```python
@router.delete("/projects/{project_id}/competitors/{competitor_id}/evidences/{evidence_id}", ...)
def delete_evidence(...):
    _require_competitor(db, project_id, competitor_id)
    ev = db.get(CompetitorEvidence, evidence_id)
    if ev is None or ev.competitor_id != competitor_id:
        raise LLMError("NOT_FOUND", "证据卡不存在")
    # 检查是否被定位引用（悬空引用保护）
    p = db.query(Positioning).filter(Positioning.project_id == project_id).first()
    if p and ev.ref_key in (json.loads(p.evidence_refs or "[]")):
        raise LLMError("CONFLICT", "该证据已被产品定位引用，请先修改定位再删除")
    db.delete(ev)
    db.commit()
    return None
```
